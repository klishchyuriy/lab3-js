function isLeap(year) {
  return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;
}

// Завдання 3
function groupByAge(people) {
  const result = {};
  for (const person of people) {
    if (!result[person.age]) {
      result[person.age] = [];
    }
    result[person.age].push(person.name);
  }
  return result;
}

const people = [
  { name: "Oleg", age: 25 },
  { name: "Alla", age: 30 },
  { name: "Peter", age: 25 },
  { name: "Ivan", age: 30 }
];

console.log(groupByAge(people));
